Step 1: One-Click Supabase Table SQL

Paste this SQL directly into your Supabase Dashboard > SQL Editor and click "RUN". It sets up the schema with public read and insert policies in 3 seconds:

-- ========================================================= -- YAFI AHMED ECOSYSTEM - PRODUCTION SUPABASE DATABASE SCHEMA -- ========================================================= -- 1. PORTFOLIO CONTENT TABLE (Main CMS Store) create table if not exists public.portfolio_content ( id text primary key default 'main', data jsonb not null, updated_at timestamp with time zone default timezone('utc'::text, now()) not null ); alter table public.portfolio_content enable row level security; drop policy if exists "Public Read Content" on public.portfolio_content; create policy "Public Read Content" on public.portfolio_content for select using (true); drop policy if exists "Anon Full Access Content" on public.portfolio_content; create policy "Anon Full Access Content" on public.portfolio_content for all using (true) with check (true); -- 2. CONTACT INQUIRIES TABLE (Client & Transmission Inbox) create table if not exists public.contact_inquiries ( id uuid primary key default gen_random_uuid(), name text not null, email text not null, subject text default 'General Inquiry', message text not null, status text default 'unread', created_at timestamp with time zone default timezone('utc'::text, now()) not null ); alter table public.contact_inquiries enable row level security; drop policy if exists "Public Can Submit Inquiries" on public.contact_inquiries; create policy "Public Can Submit Inquiries" on public.contact_inquiries for insert with check (true); drop policy if exists "Admin Can View Inquiries" on public.contact_inquiries; create policy "Admin Can View Inquiries" on public.contact_inquiries for select using (true); create index if not exists idx_inquiries_created_at on public.contact_inquiries (created_at desc); -- 3. USER PROFILES TABLE (Social OAuth & Member Database) create table if not exists public.user_profiles ( id uuid references auth.users on delete cascade primary key, full_name text, email text, phone text, avatar_url text, provider text, role text default 'member', created_at timestamp with time zone default timezone('utc'::text, now()) not null, updated_at timestamp with time zone default timezone('utc'::text, now()) not null ); alter table public.user_profiles enable row level security; drop policy if exists "Public Read User Profiles" on public.user_profiles; create policy "Public Read User Profiles" on public.user_profiles for select using (true); drop policy if exists "Users Can Update Own Profile" on public.user_profiles; create policy "Users Can Update Own Profile" on public.user_profiles for update using (auth.uid() = id); create index if not exists idx_profiles_email on public.user_profiles (email); -- 4. AUTOMATIC PROFILE TRIGGER ON SIGNUP / SOCIAL OAUTH create or replace function public.handle_new_user() returns trigger as $$ begin insert into public.user_profiles (id, full_name, email, phone, avatar_url, provider) values ( new.id, coalesce(new.raw_user_meta_data->>'full_name', new.raw_user_meta_data->>'name', split_part(new.email, '@', 1)), new.email, new.raw_user_meta_data->>'phone', coalesce(new.raw_user_meta_data->>'avatar_url', new.raw_user_meta_data->>'picture'), coalesce(new.raw_app_meta_data->>'provider', 'email') ) on conflict (id) do update set full_name = coalesce(excluded.full_name, user_profiles.full_name), avatar_url = coalesce(excluded.avatar_url, user_profiles.avatar_url), updated_at = now(); return new; end; $$ language plpgsql security definer; drop trigger if exists on_auth_user_created on auth.users; create trigger on_auth_user_created after insert on auth.users for each row execute procedure public.handle_new_user(); -- 5. GRANT PERMISSIONS grant usage on schema public to anon, authenticated; grant all on all tables in schema public to anon, authenticated; grant all on all sequences in schema public to anon, authenticated;
